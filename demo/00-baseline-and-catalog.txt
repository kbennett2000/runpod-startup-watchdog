Source of the catalog script quoted above:

    """Read GET /v2/catalog/cpus and print the flavors cheapest first.
    
    A one-off research step for the proving run, not part of the tool: the watchdog never needs the
    catalog. Kept in the scratchpad rather than the repo for that reason, and reproduced here in the
    transcript so the choice of instance is checkable.
    
    The key comes from RUNPOD_API_KEY and is never printed.
    """
    
    import os
    import sys
    
    import requests
    
    key = os.environ.get("RUNPOD_API_KEY", "").strip()
    if not key:
        sys.exit("No RUNPOD_API_KEY in the environment.")
    
    response = requests.get(
        "https://api.runpod.io/v2/catalog/cpus",
        headers={"Authorization": f"Bearer {key}", "Accept": "application/json"},
        params={"include": "AVAILABILITY"},
        timeout=(10, 30),
    )
    print(f"HTTP {response.status_code}")
    response.raise_for_status()
    
    cpus = response.json().get("cpus", [])
    print(f"{len(cpus)} CPU flavors\n")
    
    rows = []
    for cpu in cpus:
        price = (cpu.get("price") or {}).get("securePerVcpu")
        vcpu = cpu.get("vcpu") or {}
        rows.append((
            price if isinstance(price, (int, float)) else 9e9,
            cpu.get("id", ""),
            cpu.get("name", ""),
            f"{vcpu.get('min')}-{vcpu.get('max')}",
            str(price),
            str(cpu.get("availability")),
            ",".join(dc.get("id", "") for dc in (cpu.get("dataCenters") or [])),
        ))
    rows.sort()
    
    header = ("FLAVOR", "NAME", "VCPU", "USD/VCPU/HR", "AVAIL", "DATA CENTERS")
    table = [header] + [row[1:] for row in rows]
    widths = [max(len(r[i]) for r in table) for i in range(len(header))]
    for row in table:
        print("  ".join(cell.ljust(widths[i]) for i, cell in enumerate(row)).rstrip())
    
    print("\nUSD/VCPU/HR is Runpod account spend, secure cloud, per vCPU per hour.")
